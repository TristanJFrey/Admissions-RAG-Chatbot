# Optional: clean old artifacts so manifest/chunks stay in sync
Remove-Item -Recurse -Force indexes\faiss_admissions, chunks_txt

# Ingest Markdown -> FAISS index + manifest + chunk text dumps
python src/ingest_corpus.py --doc data/admissions.md --out indexes/faiss_admissions --chunk_size 1000 --chunk_overlap 150 --dump_chunks chunks_txt

# Query FAISS (dense)
python src/query_faiss.py --index indexes/faiss_admissions --q "When is the undergrad application deadline?" --k 5

# Query BM25 (lexical) using manifest to align chunks
python src/query_bm25.py --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --q "When is the undergrad application deadline?" --k 5

# Hybrid RRF (dense + lexical)
python src/query_hybrid_rrf.py --index indexes/faiss_admissions --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --q "When is the undergrad application deadline?" --k 5 --faiss_k 10 --bm25_k 10

# Evaluate retrieval
python src/eval_retrievers.py --index indexes/faiss_admissions --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --questions data/sample_questions.json --show_details --report_k 3

# Answer generation (qualitative). Default generator: Gemini; override with --gen_model for HF models.
python src/answer_hybrid_rag.py --index indexes/faiss_admissions --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --q "When is the undergrad application deadline?" --k 3 --faiss_k 8 --bm25_k 8 --max_new_tokens 500 --out evaluation/qualitative.txt

# Answer-level eval (uses same generator). Adjust k to control how many fused chunks are passed to the LLM.
# k-1
python src/eval_answers.py --index indexes/faiss_admissions --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --questions data/small_sample_questions.json --k 1 --faiss_k 3 --bm25_k 3 --max_new_tokens 64 --report_details --report_answers --out evaluation/eval_report_k1.txt

# k-2

# k-3
python src/eval_answers.py --index indexes/faiss_admissions --doc data/admissions.md --manifest indexes/faiss_admissions/chunk_manifest.json --questions data/small_sample_questions.json --k 3 --faiss_k 3 --bm25_k 3 --max_new_tokens 64 --report_details --report_answers --out evaluation/eval_report_k3.txt
